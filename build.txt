
OS = WINDOWS

# Paths
SOURCE_DIR = src
OBJECT_DIR = bin/${OS}
OUTPUT_DIR = build

WINDOWS_EXE = a.exe
LINUX_EXE   = a.out
OUTPUT = ${${OS}_EXE}

# Compiler settings
COMPILER = clang++
LINKER   = clang++
STANDARD = -std=c++11
DEFINE   =
INCLUDE  = -I"include"
WARNING  = -Wall -pedantic -Wextra -Wsign-promo -Wfloat-equal -Wconversion -Wno-unused-parameter -Wno-unused-value -Wnon-virtual-dtor -Wno-unused-function

WINDOWS_LDD_LIB = -lws2_32
LINUX_LDD_LIB   = 
LDD_LIB         = ${${OS}_LDD_LIB}


# System
COMPILE= ${COMPILER} -c ${__SOURCE__} -o ${__OBJECT__} ${INCLUDE} ${STANDARD} ${DEFINE} ${WARNING} -MMD
LINK   = ${LINKER} ${__OBJECTS__} -o ${__OUTPUT__} ${LDD_LIB}
